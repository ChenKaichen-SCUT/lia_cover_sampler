cmake_minimum_required(VERSION 3.16)
project(lia_cover_sampler VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
unset(Z3_INCLUDE_DIR CACHE)
unset(Z3_LIBRARY CACHE)
set(Z3_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../HighDiv-master/venv" CACHE PATH "Frozen HighDiv Z3 installation")
find_path(Z3_INCLUDE_DIR z3++.h PATHS "${Z3_PREFIX}/include" NO_DEFAULT_PATH REQUIRED)
find_library(Z3_LIBRARY z3 PATHS "${Z3_PREFIX}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(GMP_LIBRARY gmp REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)
add_library(lia_core src/core.cpp src/local.cpp src/global.cpp)
file(GLOB LIA_SOURCE_FILES CONFIGURE_DEPENDS src/*.cpp include/*.hpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${LIA_SOURCE_FILES})
set(LIA_SOURCE_DIGESTS "")
foreach(LIA_SOURCE_FILE IN LISTS LIA_SOURCE_FILES)
  file(SHA256 "${LIA_SOURCE_FILE}" LIA_DIGEST)
  string(APPEND LIA_SOURCE_DIGESTS "${LIA_DIGEST}")
endforeach()
string(SHA256 LIA_SOURCE_SHA "${LIA_SOURCE_DIGESTS}")
target_compile_definitions(lia_core PUBLIC LIA_VERSION="${PROJECT_VERSION}" LIA_SOURCE_SHA="${LIA_SOURCE_SHA}")
target_include_directories(lia_core PUBLIC include "${Z3_INCLUDE_DIR}")
target_link_libraries(lia_core PUBLIC "${Z3_LIBRARY}" "${GMPXX_LIBRARY}" "${GMP_LIBRARY}")
target_compile_options(lia_core PRIVATE -Wall -Wextra -Wpedantic)
set(CMAKE_BUILD_RPATH "${Z3_PREFIX}/lib")
add_executable(lia_cover_sampler src/main.cpp)
target_link_libraries(lia_cover_sampler PRIVATE lia_core)
add_executable(math_probe tests/math_probe.cpp)
target_include_directories(math_probe PRIVATE include)
target_link_libraries(math_probe PRIVATE "${GMPXX_LIBRARY}" "${GMP_LIBRARY}")
add_executable(core_test tests/core_test.cpp)
target_link_libraries(core_test PRIVATE lia_core)
enable_testing()
add_test(NAME core COMMAND core_test)
